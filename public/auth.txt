================================================================================
WAYFER API CURL COMMANDS - ADMIN AUTHENTICATION
Base URL: http://localhost:5000 (or http://localhost:5001)
================================================================================

1. REGISTER ADMIN
--------------------------------------------------------------------------------
curl -X POST http://localhost:5000/api/admin/auth/register \
  -H "Content-Type: application/json" \
  -d '{
    "name": "Super Admin",
    "email": "admin@wayfer.com",
    "phone": "+919876543210",
    "password": "AdminPassword@123",
    "role": "admin"
  }'

PowerShell:
Invoke-RestMethod -Uri "http://localhost:5000/api/admin/auth/register" -Method Post -Headers @{"Content-Type"="application/json"} -Body '{"name":"Super Admin","email":"admin@wayfer.com","phone":"+919876543210","password":"AdminPassword@123","role":"admin"}'

  "message": "Admin registered successfully",

2. LOGIN ADMIN (Email + Password)
--------------------------------------------------------------------------------
curl -X POST http://localhost:5000/api/admin/auth/login \
  -H "Content-Type: application/json" \
  -d '{
    "email": "admin@wayfer.com",
    "password": "AdminPassword@123"
  }'

PowerShell:
Invoke-RestMethod -Uri "http://localhost:5000/api/admin/auth/login" -Method Post -Headers @{"Content-Type"="application/json"} -Body '{"email":"admin@wayfer.com","password":"AdminPassword@123"}'

Sample Response (200 OK):

3. FORGOT PASSWORD (REQUEST RESET LINK / TOKEN)
--------------------------------------------------------------------------------
curl -X POST http://localhost:5000/api/admin/auth/forgot-password \
  -H "Content-Type: application/json" \
  -d '{
    "email": "admin@wayfer.com"
  }'

PowerShell:
Invoke-RestMethod -Uri "http://localhost:5000/api/admin/auth/forgot-password" -Method Post -Headers @{"Content-Type"="application/json"} -Body '{"email":"admin@wayfer.com"}'

Sample Response (200 OK):
{
  "status": "success",
  "message": "Password reset instructions sent to your email address",
  "data": {
    "resetToken": "a4f89d318e47d48375e244d2d488581f147a4ecf0bc1c27e85da3862d7c58850"
  }
}


4. RESET PASSWORD (WITH TOKEN & NEW PASSWORD)
--------------------------------------------------------------------------------
curl -X POST http://localhost:5000/api/admin/auth/reset-password \
  -H "Content-Type: application/json" \
  -d '{
    "token": "a4f89d318e47d48375e244d2d488581f147a4ecf0bc1c27e85da3862d7c58850",
    "newPassword": "NewAdminPassword@456"
  }'

PowerShell:
Invoke-RestMethod -Uri "http://localhost:5000/api/admin/auth/reset-password" -Method Post -Headers @{"Content-Type"="application/json"} -Body '{"token":"a4f89d318e47d48375e244d2d488581f147a4ecf0bc1c27e85da3862d7c58850","newPassword":"NewAdminPassword@456"}'

Sample Response (200 OK):
{
  "status": "success",
  "message": "Password has been reset successfully. Please log in with your new password."
}


5. GET CURRENT ADMIN PROFILE (VERIFY BEARER TOKEN)
--------------------------------------------------------------------------------
curl -X GET http://localhost:5000/api/admin/auth/me \
  -H "Authorization: Bearer YOUR_ADMIN_JWT_TOKEN"

PowerShell:
Invoke-RestMethod -Uri "http://localhost:5000/api/admin/auth/me" -Method Get -Headers @{ Authorization = "Bearer YOUR_ADMIN_JWT_TOKEN" }

Sample Response (200 OK):
{
  "status": "success",
  "data": {
    "admin": {
      "id": 1,
      "name": "Super Admin",
      "email": "admin@wayfer.com",
      "phone": "+919876543210",
      "role": "admin",
      "isActive": true,
      "createdAt": "2026-10-23T14:20:00.000Z"
    }
  }
}


6. REFRESH ACCESS TOKEN (With Token Rotation)
--------------------------------------------------------------------------------
curl -X POST http://localhost:5000/api/admin/auth/refresh-token \
  -H "Content-Type: application/json" \
  -d '{
    "refreshToken": "YOUR_REFRESH_TOKEN"
  }'

PowerShell:
Invoke-RestMethod -Uri "http://localhost:5000/api/admin/auth/refresh-token" -Method Post -Headers @{"Content-Type"="application/json"} -Body '{"refreshToken":"YOUR_REFRESH_TOKEN"}'

Sample Response (200 OK):
{
  "status": "success",
  "message": "Admin access token refreshed successfully",
  "data": {
    "token": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
    "accessToken": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
    "refreshToken": "f8a7b6c5d4e3029182736451a0b9c8d7e6f5a4b3c2d1e0f9a8b7c6d5e4f3a2b1",
    "expiresIn": "15m",
    "refreshExpiresAt": "2026-11-22T14:30:00.000Z"
  }
}


7. LOGOUT ADMIN (Revoke Refresh Token)
--------------------------------------------------------------------------------
curl -X POST http://localhost:5000/api/admin/auth/logout \
  -H "Content-Type: application/json" \
  -d '{
    "refreshToken": "YOUR_REFRESH_TOKEN"
  }'

PowerShell:
Invoke-RestMethod -Uri "http://localhost:5000/api/admin/auth/logout" -Method Post -Headers @{"Content-Type"="application/json"} -Body '{"refreshToken":"YOUR_REFRESH_TOKEN"}'

Sample Response (200 OK):
{
  "status": "success",
  "message": "Admin logged out successfully"
}
